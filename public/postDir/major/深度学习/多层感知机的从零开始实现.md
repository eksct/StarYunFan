from d2l import torch as d2l
import torch
from torch import nn //神经网络模块
<!--加载数据-->
batch_size = 256
train_iter, test_iter = d2l.load_data_fashion_mnist(batch_size)
<!--初始化模型参数-->
 <!--每张图片展开后的像素数量。10 是类别数。256 是隐藏层的大小，也就是隐藏层有 256 个神经元。-->
num_inputs, num_outputs, num_hiddens = 784, 10, 256
<!--第一层的权重矩阵。形状是 (784, 256)，意思是把 784 维的输入映射到 256 维的隐藏层。torch.randn 生成随机数，乘以 0.01 让初始值很小，避免一开始就产生过大的输出。-->
<!--requires_grad=True 告诉 PyTorch：“这个参数需要计算梯度”，训练时才会更新它。-->
W1 = nn.Parameter(torch.randn(num_inputs, num_hiddens, requires_grad=True) * 0.01)
<!--1层的偏置量0填充-->
b1 = nn.Parameter(torch.zeros(num_hiddens, requires_grad=True))
<!--第二层的权重举证，形状是 (256, 10)，把 256 维的隐藏层映射到 10 维的输出（对应 10 个类别）。-->
W2 = nn.Parameter(torch.randn(num_hiddens, num_outputs, requires_grad=True) * 0.01)
<!--2层的偏置量10填充-->
b2 = nn.Parameter(torch.zeros(num_outputs, requires_grad=True))

params = [W1, b1, W2, b2]
<!--自定义relu激活函数-->
<!--relu：输入是正数就原样保留，输入是负数就变成0。-->
def relu(X):
    return torch.max(input=X, other=torch.tensor(0.0))
<!--定义模型-->
def net(X):
<!--把每张图片从 28×28 的二维矩阵“拉平”成长度 784 的一维向量。-1 表示“这个维度的大小让 PyTorch 自己算”，实际上就是批量大小。比如一批 256 张图片，变换后 X 的形状就是 (256, 784)。-->
    X = X.view((-1, num_inputs))
<!--这是第一层的计算。torch.matmul(X, W1) 做矩阵乘法，把 (256, 784) 的 X 和 (784, 256) 的 W1 相乘，得到 (256, 256) 的结果；再加上偏置 b1；最后经过 ReLU 激活函数。得到的 H 就是隐藏层的输出，形状是 (256, 256)。-->
    H = relu(torch.matmul(X, W1) + b1)
<!--这是第二层的计算。把隐藏层输出 H 和 W2 相乘，得到 (256, 10) 的结果，再加上 b2。最终输出是一个 (256, 10) 的张量，每一行代表一张图片属于 10 个类别的“得分”。-->
    return torch.matmul(H, W2) + b2
<!--nn.CrossEntropyLoss() 是 PyTorch 提供的交叉熵损失函数。它内部会自动做 softmax（把输出变成概率分布），然后计算交叉熵损失。-->
loss = nn.CrossEntropyLoss()

num_epochs, lr = 10，0.1
d2l.train_ch3(net, train_iter, test_iter, loss, num_epochs, 
              torch.optim.SGD, lr, params)
